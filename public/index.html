import express from 'express'; import path from 'node:path'; import { fileURLToPath } from 'node:url'; const __dirname = path.dirname(fileURLToPath(import.meta.url)); const app = express(); app.use(express.json({ limit: '1mb' })); app.use(express.static(path.join(__dirname, 'public'))); const MODEL = process.env.GROQ_MODEL || 'openai/gpt-oss-20b'; app.get('/api/health', (_req, res) => { res.json({ ok: true, aiConfigured: Boolean(process.env.GROQ_API_KEY), provider: 'Groq', model: MODEL }); }); app.post('/api/chat', async (req, res) => { try { if (!process.env.GROQ_API_KEY) { return res.status(503).json({ error: 'AI is not configured. Add GROQ_API_KEY in Render.' }); } const { messages, language } = req.body ?? {}; if (!Array.isArray(messages) || messages.length === 0) { return res.status(400).json({ error: 'Please send a question.' }); } const safeMessages = messages.slice(-20).map(m => ({ role: m.role === 'assistant' ? 'assistant' : 'user', content: String(m.content ?? '').slice(0, 8000) })); const langInstruction = language === 'hi' ? 'Reply in clear, natural Hindi. Keep technical terms in English where useful.' : language === 'en' ? 'Reply in clear, accessible English.' : 'Reply in the language used by the student.'; const systemMessage = { role: 'system', content: `You are Gem, a helpful AI education tutor. Your name is Gem – General Education Mate. Help students with: - UPSC preparation - SSC examinations - Banking examinations - Railway examinations - State PCS - General Studies - History and Geography - Indian Polity - Science and Technology - Mathematics and Reasoning - Current Affairs - School and college subjects Give accurate, detailed and easy-to-understand answers. For MCQs, provide four options, the correct answer and an explanation. Explain difficult concepts step by step. Use examples wherever helpful. Use headings and bullet points for readability. Never invent facts or pretend to have live information. If unsure, clearly say so. ${langInstruction}` }; const upstream = await fetch( 'https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Authorization': `Bearer ${process.env.GROQ_API_KEY}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ model: MODEL, messages: [systemMessage, ...safeMessages], temperature: 0.6, max_completion_tokens: 2048 }) } ); const data = await upstream.json(); if (!upstream.ok) { console.error('Groq API error:', data); return res.status(502).json({ error: data?.error?.message || `Groq returned HTTP ${upstream.status}` }); } const answer = data.choices?.[0]?.message?.content; if (!answer) { return res.status(502).json({ error: 'Gem could not generate an answer. Please try again.' }); } res.json({ answer }); } catch (err) { console.error('Gem chat error:', err); res.status(500).json({ error: 'Something went wrong. Please try again.' }); } }); const port = Number(process.env.PORT || 3000); app.listen(port, () => { console.log(`Gem AI running on port ${port}`); });
